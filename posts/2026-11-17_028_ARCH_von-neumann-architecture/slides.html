mem = [("LOAD", 4), ("ADD", 5), ("STORE", 6),
       ("HALT", 0), 10, 20, 0]
pc = acc = 0
while True:
    op, a = mem[pc]; pc += 1  # 명령어도 mem
    if op == "LOAD": acc = mem[a] # 데이터도 mem
    elif op == "ADD": acc += mem[a]
    elif op == "STORE": mem[a] = acc
    else: break
print(mem[6])  # 30
① 0~3번지 명령어, 4~6번지 데이터 — 같은 mem
