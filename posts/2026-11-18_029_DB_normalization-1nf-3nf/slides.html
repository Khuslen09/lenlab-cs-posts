SQL로 만든 3NF
3NF tables in SQL · SQL дээрх 3NF
CREATE TABLE dept (
  dept TEXT PRIMARY KEY, office TEXT);
CREATE TABLE student (
  sid INT PRIMARY KEY, name TEXT,
  dept TEXT REFERENCES dept(dept));
CREATE TABLE enroll (
  sid INT REFERENCES student(sid),
  course TEXT, grade TEXT,
  PRIMARY KEY (sid, course));
① 사무실 이전 → dept 1행만 수정
